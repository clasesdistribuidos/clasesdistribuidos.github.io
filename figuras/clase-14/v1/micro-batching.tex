% Lotes de un minuto sobre la línea de tiempo (los cortes rojos, como en la
% pizarra); el evento de las 8:00:05 llega cuando el primer lote ya se cerró.
\begin{document}
\begin{figura}\begin{tikzpicture}
  \draw[->] (0,0) -- (8.6,0);
  \foreach \i in {0,...,6}
    \draw[acento, dashed, line width=0.6pt] ({0.3+1.3*\i},-0.1) -- ({0.3+1.3*\i},4.2);
  \foreach \x/\t in {0.3/8:00, 1.6/8:01}
    \draw (\x,-0.12) -- (\x,0.12) node[etiq, below=6pt] {\t};
  \foreach \p in {(0.9,1.0), (2.1,2.3), (3.0,1.2), (3.7,2.9), (4.2,1.8), (5.3,1.0), (6.9,1.4), (7.4,2.8)}
    \node[evento] at \p {};
  \node[evento, minimum size=6.5pt] (t) at (6.2,2.6) {};
  \node[etiq, below=3pt of t] {8:00:05};
  \draw[->] (t) to[out=110, in=60, looseness=0.9] (1.0,2.9);
\end{tikzpicture}\end{figura}
\end{document}
