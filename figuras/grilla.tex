% Grilla de referencia para el PDF de revisión (render_figuras.py --revision).
% Va detrás del dibujo, en cm absolutos y numerada en los bordes, para que las
% correcciones a mano puedan decir "subilo a la altura 2". Nunca llega al sitio.
\usetikzlibrary{backgrounds}
\newcommand\grilla{%
  \pgfpointanchor{current bounding box}{south west}\pgfgetlastxy\gxa\gya
  \pgfpointanchor{current bounding box}{north east}\pgfgetlastxy\gxb\gyb
  \pgfmathtruncatemacro\gxi{floor(\gxa/28.45274)}\pgfmathtruncatemacro\gxf{ceil(\gxb/28.45274)}
  \pgfmathtruncatemacro\gyi{floor(\gya/28.45274)}\pgfmathtruncatemacro\gyf{ceil(\gyb/28.45274)}
  \begin{scope}[x=1cm, y=1cm, on background layer]
    \draw[black!10, line width=0.2pt, step=0.5] (\gxi,\gyi) grid (\gxf,\gyf);
    \draw[black!22, line width=0.3pt, step=1] (\gxi,\gyi) grid (\gxf,\gyf);
    \foreach \x in {\gxi,...,\gxf} \node[font=\scriptsize, text=black!45, below] at (\x,\gyi) {\x};
    \foreach \y in {\gyi,...,\gyf} \node[font=\scriptsize, text=black!45, left] at (\gxi,\y) {\y};
  \end{scope}}
\tikzset{every picture/.append style={execute at end picture={\grilla}}}
