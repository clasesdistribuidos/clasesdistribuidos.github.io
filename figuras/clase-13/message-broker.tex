% Propuesta: la misma comparación que mensajeria-directa, con los mismos dos
% productores y tres consumidores en el mismo lugar en los dos paneles. Sin
% broker, cada productor se conecta con cada consumidor; con broker, nadie se
% conecta con nadie más que con él, y el broker es un sistema aparte, él mismo
% distribuido (los tres servidores adentro). Los servidores y los pies son
% agregados; salen del texto ("los nodos no se conectan entre sí, sino que
% existe un sistema aparte"; "las dos versiones son distribuidas").
\begin{document}
\begin{figura}\begin{tikzpicture}[
    q/.style={caja, minimum width=11mm, minimum height=10mm},
  ]
  \newcommand\extremos{%
    \foreach \i/\y in {1/0.9, 2/-0.9} \node[q] (p\i) at (0,\y) {\var{P}};
    \foreach \i/\y in {1/1.7, 2/0, 3/-1.7} \node[q] (c\i) at (5.2,\y) {\var{C}};}
  \newcommand\pie[2]{\node[etiq, text=suave] at (#1,-2.75) {#2};}
  \newcommand\titulo[2]{\node at (#1,-3.3) {#2};}

  % --- mensajería directa
  \extremos
  \foreach \p in {p1, p2} \foreach \c in {c1, c2, c3} \draw[msg] (\p.east) -- (\c.west);
  \pie{2.6}{cada productor conoce a cada consumidor}
  \titulo{2.6}{mensajería directa}

  % --- message broker
  \begin{scope}[xshift=8.4cm]
    \extremos
    \draw[marco, fill=relleno] (1.8,-1.0) rectangle (3.4,1.0);
    \foreach \y in {0.5, 0, -0.5} \node[componente] at (2.6,\y) {};
    \node[anchor=south] (b) at (2.6,1.0) {broker};
    \foreach \i/\y in {1/0.9, 2/-0.9} \draw[msg] (p\i.east) -- (1.8,{0.45*\y});
    \foreach \c in {c1, c2, c3} \draw[msg] (3.4,0) -- (\c.west);
    \pie{2.6}{cada uno conoce solo al broker}
    \titulo{2.6}{message broker}
  \end{scope}
\end{tikzpicture}\end{figura}
\end{document}
